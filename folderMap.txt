{ config, pkgs, ... }: {
  nixpkgs.overlays = [
    (final: prev: {
      gpuWrap = pkg:
        # Check if the host has PRIME offload explicitly enabled
        if (config.hardware.nvidia.prime.offload.enable or false) then
          prev.symlinkJoin {
            name = "${pkg.name}-gpu";
            paths = [ pkg ];
            buildInputs = [ prev.makeWrapper ];
            postBuild = ''
              # Iterate through common executable directories
              for dir in "$out/bin" "$out/libexec"; do
                if [ -d "$dir" ]; then
                  for target in "$dir"/*; do
                    # Only apply wrapProgram if the target is an executable file
                    if [ -f "$target" ] && [ -x "$target" ]; then
                      wrapProgram "$target" \
                        --set __NV_PRIME_RENDER_OFFLOAD 1 \
                        --set __VK_LAYER_NV_optimus NVIDIA_only \
                        --set __GLX_VENDOR_LIBRARY_NAME nvidia
                    fi
                  done
                fi
              done
            '';
          }
        else
          # Fallback: return the application untouched for the VM and Server
          pkg;
    })
  ];
}

{ pkgs, ... }:

let
  # Shared helper interpolated into the scripts that push to GitHub
  retryGitPush = ''
    _retry_git_push() {
      if git push origin main; then
        return 0
      else
        echo -e "\nERROR: git push failed. (Likely the Wayland SSH agent refusing operation)"
        read -p "Would you like to secure the key, authorize it via CLI, and retry? (y/N): " retry_ans
        if [[ "$retry_ans" =~ ^[Yy]$ ]]; then
          echo "1. Securing key permissions (chmod 600)..."
          chmod 600 ~/.ssh/id_ed25519

          echo "2. Forcing manual passphrase prompt..."
          ssh-add ~/.ssh/id_ed25519

          echo "3. Verifying GitHub access..."
          if ssh -T git@github.com 2>&1 | grep -q "successfully authenticated"; then
            echo "GitHub authentication successful! Retrying push..."
            git push origin main
            return $?
          else
            echo "GitHub authentication failed. You will need to check your keys manually."
            return 1
          fi
        else
          echo "Skipping push. You will need to manually push later."
          return 1
        fi
      fi
    }
  '';

  nix-test = pkgs.writeShellScriptBin "nix-test" ''
    host=$(hostname)
    pushd ~/.nixos > /dev/null
    git add -A
    echo "Testing local configuration for $host..."
    sudo nixos-rebuild test --flake ".#$host"
    popd > /dev/null
  '';

  nix-test-back = pkgs.writeShellScriptBin "nix-test-back" ''
    host=$(hostname)
    pushd ~/.nixos > /dev/null
    echo "WARNING: Scrapping uncommitted changes..."
    git reset --hard HEAD
    git clean -fd

    echo "Reverting live system to the last commit..."
    sudo nixos-rebuild switch --flake ".#$host"
    popd > /dev/null
  '';

  nix-deploy = pkgs.writeShellScriptBin "nix-deploy" ''
    ${retryGitPush}
    msg="chore: update system configuration"
    host=$(hostname)
    if [[ "$1" == "-m" && -n "$2" ]]; then
      msg="$2"
    fi
    pushd ~/.nixos > /dev/null

    git add -A

    echo "Building local configuration..."
    if sudo nixos-rebuild build --flake ".#$host"; then
      echo "Build successful! Activating live system..."

      if sudo nixos-rebuild switch --flake ".#$host"; then
        if ! git diff --cached --quiet; then
          git commit -m "$msg"
          git pull origin main --rebase --autostash || true
          if _retry_git_push; then
            echo "Build successful. Known-good configuration pushed to GitHub."
          else
            echo "ERROR: Build and commit successful, but git push failed! Configuration is only local."
          fi
        else
          echo "Build successful. No git changes to commit."
        fi
      else
        echo "Live activation failed (likely a service issue). Falling back to boot..."
        sudo nixos-rebuild boot --flake ".#$host"

        if ! git diff --cached --quiet; then
          git commit -m "$msg (requires reboot)"
          git pull origin main --rebase --autostash || true
          if _retry_git_push; then
            echo "Configuration committed and pushed. Please reboot your machine to apply the changes."
          else
            echo "ERROR: Configuration committed locally, but git push failed! Please reboot and manually push later."
          fi
        fi
      fi
    else
      echo "Build failed! Code error detected. Nothing was pushed to GitHub."
    fi
    popd > /dev/null
  '';

  nix-deploy-back = pkgs.writeShellScriptBin "nix-deploy-back" ''
    ${retryGitPush}
    host=$(hostname)
    pushd ~/.nixos > /dev/null
    echo "Reverting the last commit..."
    git revert --no-edit HEAD

    echo "Applying reverted configuration..."
    if sudo nixos-rebuild switch --flake ".#$host"; then
      if _retry_git_push; then
        echo "Rollback successful and pushed to GitHub."
      else
        echo "ERROR: Rollback applied locally, but git push failed!"
      fi
    else
      echo "Rollback failed. You may need to fix conflicts manually."
    fi
    popd > /dev/null
  '';

  nix-upgrade = pkgs.writeShellScriptBin "nix-upgrade" ''
    ${retryGitPush}
    msg="chore: upgrade all packages (bump flake.lock)"
    host=$(hostname)
    action="switch"

    if [[ "$1" == "--boot" ]]; then
      action="boot"
    elif [[ "$1" == "-m" && -n "$2" ]]; then
      msg="$2"
    fi

    pushd ~/.nixos > /dev/null
    echo "Fetching newest packages from Nixpkgs..."
    nix flake update

    echo "Building the new generation..."
    if sudo nixos-rebuild build --flake ".#$host"; then
      echo "Build successful! Attempting system $action..."

      if sudo nixos-rebuild "$action" --flake ".#$host"; then
        git add flake.lock
        if ! git diff --cached --quiet; then
          git commit -m "$msg"
          git pull origin main --rebase --autostash || true
          if _retry_git_push; then
            echo "System upgraded successfully. New lockfile pushed to GitHub."
          else
            echo "ERROR: System upgraded and lockfile committed locally, but git push failed!"
          fi
        else
          echo "Packages are already up to date."
        fi
      else
        echo "Live activation failed. Falling back to boot..."
        sudo nixos-rebuild boot --flake ".#$host"

        git add flake.lock
        if ! git diff --cached --quiet; then
          git commit -m "$msg (requires reboot)"
          git pull origin main --rebase --autostash || true
          if _retry_git_push; then
            echo "Lockfile committed and pushed. Please reboot to apply the upgrade."
          else
            echo "ERROR: Lockfile committed locally, but git push failed! Please reboot and push manually."
          fi
        fi
      fi
    else
      echo "Upgrade compilation failed! Rolling back flake.lock..."
      git checkout flake.lock
    fi
    popd > /dev/null
  '';

  nix-clean = pkgs.writeShellScriptBin "nix-clean" ''
      echo "1. Deploying current configuration..."
      if ! nix-deploy "$@"; then
        echo "ERROR: nix-deploy failed. Aborting cleanup to preserve rollback generations."
        exit 1
      fi

      echo "2. Deleting old system and user generations + collecting garbage..."
      sudo nix-collect-garbage -d
      nix-collect-garbage -d

      echo "3. Resetting current generation to #1..."
      current_target=$(readlink -f /nix/var/nix/profiles/system)
      current_link=$(readlink /nix/var/nix/profiles/system)

      if [[ "$current_link" != "system-1-link" ]]; then
        sudo ln -sfn "$current_target" /nix/var/nix/profiles/system-1-link
        sudo ln -sfn system-1-link /nix/var/nix/profiles/system
        sudo rm -f "/nix/var/nix/profiles/$current_link"
      fi

      echo "4. Updating boot menu..."
      sudo /run/current-system/bin/switch-to-configuration boot

      echo "5. Current system generations:"
      sudo nix-env --list-generations --profile /nix/var/nix/profiles/system
    '';
in
{
  environment.systemPackages = [
    nix-test
    nix-test-back
    nix-deploy
    nix-deploy-back
    nix-upgrade
    nix-clean
  ];
}

----------------------------------------
Location: modules/programs/rclone.nix

{ config, pkgs, ... }:

{
  environment.systemPackages = [
    pkgs.rclone

    # --- SETUP SCRIPT ---
    (pkgs.writeShellScriptBin "nix-rclone-setup" ''
      #!/usr/bin/env bash
      set -e

      REMOTE=""
      DIR_NAME=""
      CLOUD_DIR=""

      # Parse command line flags
      while [[ $# -gt 0 ]]; do
        case $1 in
          -r|--remote) REMOTE="$2"; shift 2 ;;
          -d|--dir) DIR_NAME="$2"; shift 2 ;;
          -c|--cloud-dir) CLOUD_DIR="$2"; shift 2 ;;
          *) echo "Unknown parameter: $1"; exit 1 ;;
        esac
      done

      if [ -z "$REMOTE" ] || [ -z "$DIR_NAME" ]; then
        echo "Usage: nix-rclone-setup --remote <remotename> --dir <localfolder> [--cloud-dir <remotefolder>]"
        echo "Example 1 (Whole drive): nix-rclone-setup --remote onedrive --dir AllFiles"
        echo "Example 2 (Specific folder): nix-rclone-setup --remote onedrive --dir School --cloud-dir Documents/School"
        exit 1
      fi

      # Construct the remote path depending on whether a cloud dir was provided
      if [ -z "$CLOUD_DIR" ]; then
        REMOTE_PATH="$REMOTE:"
      else
        REMOTE_PATH="$REMOTE:$CLOUD_DIR"
      fi

      BASE_DIR="$HOME/Rclone"
      SYNC_DIR="$BASE_DIR/$DIR_NAME"
      ENV_DIR="$HOME/.config/rclone"
      ENV_FILE="$ENV_DIR/bisync-$DIR_NAME.env"
      WANTS_DIR="$HOME/.config/systemd/user/timers.target.wants"

      mkdir -p "$SYNC_DIR" "$ENV_DIR" "$WANTS_DIR"

      echo "1. Checking authentication for '$REMOTE'..."
      if ! ${pkgs.rclone}/bin/rclone listremotes | grep -q "^$REMOTE:"; then
        echo "Remote '$REMOTE' not found. Launching interactive setup..."
        ${pkgs.rclone}/bin/rclone config
      fi

      echo "2. Checking bisync cache status for $DIR_NAME..."
      if [ ! -f "$ENV_FILE" ]; then
        echo "First run detected. Performing initial --resync. This may take a while..."
        ${pkgs.rclone}/bin/rclone bisync "$REMOTE_PATH" "$SYNC_DIR" --resync --verbose

        # Write environment variables for the systemd template to read
        echo "REMOTE_PATH=$REMOTE_PATH" > "$ENV_FILE"
        echo "SYNC_DIR=$DIR_NAME" >> "$ENV_FILE"
      else
        echo "Bisync environment already exists. Skipping --resync."
      fi

      echo "3. Starting background sync service for $DIR_NAME..."
      # Symlink directly to /etc/systemd/user/ so GC never breaks the link
      ln -sf "/etc/systemd/user/rclone-bisync@.timer" "$WANTS_DIR/rclone-bisync@$DIR_NAME.timer"
      systemctl --user daemon-reload
      systemctl --user start "rclone-bisync@$DIR_NAME.timer"

      echo "Setup complete! '$REMOTE_PATH' is actively syncing to '$SYNC_DIR'."
    '')

    # --- REMOVE SCRIPT ---
    (pkgs.writeShellScriptBin "nix-rclone-remove" ''
      #!/usr/bin/env bash
      set -e

      DIR_NAME=""
      REMOVE_ALL=false
      WANTS_DIR="$HOME/.config/systemd/user/timers.target.wants"

      # Parse command line flags
      while [[ $# -gt 0 ]]; do
        case $1 in
          -d|--dir) DIR_NAME="$2"; shift 2 ;;
          --all) REMOVE_ALL=true; shift 1 ;;
          *) echo "Unknown parameter: $1"; exit 1 ;;
        esac
      done

      if [ "$REMOVE_ALL" = true ]; then
        echo "Removing ALL rclone bisync timers..."
        for env_file in "$HOME/.config/rclone"/bisync-*.env; do
          [ -e "$env_file" ] || continue
          name=$(basename "$env_file" .env)
          dir_name=''${name#bisync-}

          systemctl --user stop "rclone-bisync@$dir_name.timer" 2>/dev/null || true
          rm -f "$WANTS_DIR/rclone-bisync@$dir_name.timer" \
                "$HOME/.config/systemd/user/rclone-bisync@$dir_name.timer" \
                "$env_file"
          echo "✅ Disabled sync for: $dir_name"
        done
        systemctl --user daemon-reload
        echo "All background syncs stopped. Your local files in ~/Rclone/ were NOT deleted."
        exit 0
      fi

      if [ -z "$DIR_NAME" ]; then
        echo "Usage: nix-rclone-remove --dir <foldername> OR nix-rclone-remove --all"
        exit 1
      fi

      echo "Stopping background sync for $DIR_NAME..."
      systemctl --user stop "rclone-bisync@$DIR_NAME.timer" 2>/dev/null || true
      rm -f "$WANTS_DIR/rclone-bisync@$DIR_NAME.timer" \
            "$HOME/.config/systemd/user/rclone-bisync@$DIR_NAME.timer" \
            "$HOME/.config/rclone/bisync-$DIR_NAME.env"
      systemctl --user daemon-reload

      echo "✅ Removed background sync for $DIR_NAME."
      echo "Note: Your local files are kept at ~/Rclone/$DIR_NAME."
    '')
  ];

  # --- SYSTEMD SERVICE TEMPLATE ---
  systemd.user.services."rclone-bisync@" = {
    description = "Automated Rclone Bisync for %i";
    after = [ "network-online.target" ];
    unitConfig = {
      ConditionPathExists = "%h/.config/rclone/bisync-%i.env";
    };
    serviceConfig = {
      Type = "oneshot";
      EnvironmentFile = "%h/.config/rclone/bisync-%i.env";
      ExecStart = "${pkgs.rclone}/bin/rclone bisync \${REMOTE_PATH} %h/Rclone/\${SYNC_DIR} --verbose --conflict-resolve newer --resilient --recover";
    };
  };

  # --- SYSTEMD TIMER TEMPLATE ---
  systemd.user.timers."rclone-bisync@" = {
    description = "Timer for automated rclone bisync (%i)";
    timerConfig = {
      OnBootSec = "3m";
      OnUnitActiveSec = "5m";
      Unit = "rclone-bisync@%i.service";
    };
  };
}

# .nixos

A declarative, modular NixOS configuration featuring the Umbriel compositor, Noctalia Shell, Disko-managed LUKS + Btrfs, and a streamlined GPU offload architecture.

This repository relies on a profile-based module system, allowing seamless deployment across high-performance workstations, virtual machines, and a foundational architecture for a future multi-node home server environment.

---

## 🏗️ System Architecture

To maintain a clean and reproducible system, the configuration has been fully modularized away from strict tiers into a flexible profile system. Configurations are assembled from modular building blocks:

* **Profiles (`modules/profiles/`):** The top-level definitions for specific hardware roles, including `workstation.nix`, `server.nix`, and `vm.nix`.


* **Core (`modules/core/`):** Global, OS-level configurations including bootloaders, networking, Tailscale, Btrfs scrubbing, and custom GitOps bash utilities.


* **Hardware & Desktop:** Specialized modules handling audio, Bluetooth, power management, and the Wayland environment (Umbriel + Noctalia).


* **Programs (`modules/programs/`):** Categorized application suites (gaming, creative, development, productivity).


* **Fallbacks (`modules/fallbacks/`):** Containerized environments managed declaratively via Flatpak and Distrobox/Podman, alongside the global development shell flake.



---

## 🛠️ The GitOps Workflow (Local-First)

System states are managed via custom bash commands embedded in `modules/core/nix-bash.nix`. This enforces a strict "Local-First" rule: changes are built and applied locally, and *only* pushed to the repository if the build succeeds.

| Command | Action |
| --- | --- |
| `nix-test` | Builds the system locally. Changes revert on the next reboot. Perfect for testing breaking changes. |
| `nix-test-back` | Scraps uncommitted changes and reverts the live system to the last commit. |
| `nix-deploy -m "msg"` | Builds the system locally. If successful, automatically commits and pushes the configuration. |
| `nix-deploy-back` | Reverts the last commit, applies the configuration, and pushes the rollback. |
| `nix-upgrade -m "msg"` | Fetches the latest Nixpkgs, applies them locally, and pushes the new `flake.lock` upon success. |

---

## 🔬 Development Environments (Direnv & Flakes)

Project-specific, isolated toolchains (Python, Julia, Rust, LaTeX, C/C++) are deployed dynamically on a per-folder basis using `direnv` and `nix-ld`, heavily optimized for high-performance astrodynamics computations and space engineering research.

The global flake in `modules/fallbacks/flake.nix` provides dedicated shells:

* **`python` / `jupyter**`: Core scientific Python environments (NumPy, SciPy, Pandas, SpiceyPy).


* **`rust` / `c-cpp**`: High-performance systems development toolchains.


* **`latex`**: Full TeX Live scheme with `texlab` for academic authoring.


* **`jupyter-julia`**: The ultimate astrodynamics shell, integrating Jupyter and Julia within a localized project namespace.



To activate an environment, echo the desired shell into an `.envrc` file and allow it:

```bash
echo "use flake ~/.nixos/modules/fallbacks#jupyter-julia" > .envrc
direnv allow

```

---

## 🎮 Gaming & Dedicated GPUs (PRIME)

This configuration universally applies GameMode, Gamescope, and specific input Udev rules (like 8BitDo Ultimate Controller support) via `modules/programs/gaming.nix`.

### The `gpuWrap` Overlay

Instead of manually crafting `symlinkJoin` wrappers for individual applications, this repository introduces a global `gpuWrap` overlay in `modules/core/gpu-wrapper.nix`. This automatically injects the required `__NV_PRIME_RENDER_OFFLOAD` and `__GLX_VENDOR_LIBRARY_NAME` environment variables into any wrapped package.

To force a standalone application (like Blender, FreeCAD, or SuperTuxKart) to permanently use the discrete GPU, simply call the wrapper in your package list:

```nix
environment.systemPackages = with pkgs; [
  (pkgs.gpuWrap blender)
  (pkgs.gpuWrap freecad)
  (pkgs.gpuWrap supertuxkart)
];

```

If PRIME offload is disabled on a specific host (e.g., a server or VM), the overlay intelligently ignores the wrapper and returns the standard package.

### Game Launchers

* **Steam Games**: Right-click the game in Steam -> **Properties...** -> **General** -> **Launch Options**:
```bash
gamemoderun nvidia-offload %command%

```


*(Or with MangoHud: `gamemoderun mangohud nvidia-offload %command%`)*

## 🚀 Scenario A: Reinstalling an Existing Host

Use this method if the machine's profile (like Cyron or VM) and `hardware.nix` already exist in the repository.

### Step 1: Boot & Format (Disko)

Boot the NixOS Minimal ISO and connect to the internet (use `nmtui` for Wi-Fi). Run Disko directly from GitHub to partition and mount the drives:

```bash
sudo nix --experimental-features "nix-command flakes" run github:nix-community/disko -- --mode disko --flake "github:hkn-alp/.nixos#Cyron"

```

*(Enter your chosen LUKS encryption password when prompted).*

### Step 2: Install NixOS

Since the hardware configuration is already tracked, install directly from the remote flake:

```bash
sudo nixos-install --flake "github:hkn-alp/.nixos#Cyron" --no-root-passwd
sudo reboot

```

### Step 3: Post-Installation Setup

Log in and clone your repository locally to manage future updates via the GitOps workflow:

```bash
git clone https://github.com/hkn-alp/.nixos.git ~/.nixos

```

---

## 🚀 Scenario B: Bootstrapping a New Machine

*Use this method if you are installing NixOS on a brand new computer that needs a new `hardware.nix` generated.*

### Step 1: Prep the Live Environment & Clone

Boot the NixOS Minimal ISO, connect to the internet, and spawn a temporary shell with Git:

```bash
nix-shell -p git
git clone https://github.com/hkn-alp/.nixos.git .nixos
cd .nixos

```

### Step 2: Create the Host Profile

Duplicate an existing host folder to act as your template:

```bash
cp -r hosts/Cyron hosts/NewHost

```

**1. Route the Flake**
Open `flake.nix` and add your `NewHost` to the `nixosConfigurations` block:

```nix
NewHost = nixpkgs.lib.nixosSystem {
  system = "x86_64-linux"; 
  specialArgs = { inherit inputs; };
  modules = [ ./hosts/NewHost/default.nix ];
};

```

**2. Update the Hostname**
Open `hosts/NewHost/default.nix` and ensure the hostname matches:

```nix
networking.hostName = "NewHost";

```

**3. Target the Correct Drive with Disko**
List available block devices (`lsblk`). Open `hosts/NewHost/disko.nix` and update the device target:

```nix
disko.devices.disk.main = {
  device = "/dev/YOUR_TARGET_DRIVE"; 
};

```

**4. Stage the Hardware Profile**
Create an empty placeholder for your hardware configuration and track it, otherwise the Nix installer will ignore it:

```bash
touch hosts/NewHost/hardware.nix
echo "{ ... }: {}" > hosts/NewHost/hardware.nix 
git add .

```

### Step 3: Format the Disk (Disko)

Run Disko using your *local* flake to format and mount the drives to `/mnt`:

```bash
sudo nix --experimental-features "nix-command flakes" run github:nix-community/disko -- --mode disko --flake .#NewHost

```

### Step 4: Generate Hardware Config

Generate the hardware profile:

```bash
sudo nixos-generate-config --root /mnt --dir /tmp
cp /tmp/hardware-configuration.nix hosts/NewHost/hardware.nix

```

*Crucial:* Open `hosts/NewHost/hardware.nix` and delete all `fileSystems`, `swapDevices`, and `boot.initrd.luks` blocks, as Disko handles these.

### Step 5: Install & Persist

Stage the newly generated hardware configuration in git:

```bash
git add .

```

Install the system from the local flake:

```bash
sudo nixos-install --flake .#NewHost --no-root-passwd

```

Copy the local repository directly into your new persistent home directory so it survives the reboot:

```bash
sudo cp -r .nixos /mnt/home/hakanalp/
sudo chown -R 1000:1000 /mnt/home/hakanalp/.nixos

```

### Step 6: Reboot & Push

```bash
sudo reboot

```

When the machine restarts, log in, navigate to `~/.nixos`, commit your new hardware configuration, and push it to your repository. Future updates can now be handled via `nix-deploy`.
